# ── 17.0 Path helper (defined inline) ────────────────────────────────
import os, json
from pathlib import Path
from datetime import datetime


class RunPaths:
    """Structured output paths for one validation run."""

    def __init__(self, dataset, gamma_tag, fold, sample_idx, class_name,
                 ckpt_path="", root="figures/validation"):
        self.dataset    = dataset.lower()
        self.gamma_tag  = str(gamma_tag).replace(".", "")
        self.fold       = int(fold)
        self.sample_idx = int(sample_idx)
        self.class_name = class_name.lower().replace(" ", "_")
        self.ckpt_path  = ckpt_path

        self.gamma_dir = Path(root) / self.dataset / f"gamma_{self.gamma_tag}"
        self.gamma_dir.mkdir(parents=True, exist_ok=True)

        self.image_dir = self.gamma_dir / f"fold{self.fold}_idx{self.sample_idx}_{self.class_name}"
        self.image_dir.mkdir(parents=True, exist_ok=True)

        self._tables = self.gamma_dir / "tables"
        self._tables.mkdir(parents=True, exist_ok=True)

        readme = self.gamma_dir / "README.txt"
        if not readme.exists():
            readme.write_text(
                f"ESPAOCL validation outputs — {self.dataset.upper()} γ={self.gamma_tag}\n"
                f"{'='*60}\n\n"
                f"Folder layout:\n"
                f"  fold{{F}}_idx{{N}}_{{class}}/\n"
                f"    01_BEFORE_*.png            before-alignment artefacts\n"
                f"    02_BEFORE_*_concepts.png   before-alignment concept grid\n"
                f"    03_AFTER_*.png             after-alignment artefacts\n"
                f"    04_AFTER_*_concepts.png   after-alignment concept grid\n"
                f"    05_*.png                   diagnostic figures\n"
                f"    metadata.json              prediction, score, top concepts\n"
                f"  tables/\n"
                f"    concept_table_PRE.csv\n"
                f"    concept_table_POST.csv\n"
                f"    localisation_metrics.csv\n"
            )

    def _path(self, n, stage, kind, ext="png"):
        return str(self.image_dir / f"{n:02d}_{stage}_{kind}.{ext}")

    def before_single(self):     return self._path(1, "BEFORE", "alignment_single")
    def before_concepts(self):   return self._path(2, "BEFORE", "alignment_single")
    def after_single(self):      return self._path(3, "AFTER",  "alignment_single")
    def after_concepts(self):    return self._path(4, "AFTER",  "alignment_single")
    def localisation_diag(self): return self._path(5, "AFTER",  "localisation_diagnostic")

    def tables_dir(self):  return str(self._tables)
    def table_pre(self):   return str(self._tables / "concept_table_PRE.csv")
    def table_post(self):  return str(self._tables / "concept_table_POST.csv")
    def table_loc(self):   return str(self._tables / "localisation_metrics.csv")

    def save_metadata(self, info):
        base = {
            "timestamp":       datetime.now().isoformat(timespec="seconds"),
            "dataset":         self.dataset,
            "gamma_tag":       self.gamma_tag,
            "fold":            self.fold,
            "sample_idx":      self.sample_idx,
            "class_name":      self.class_name,
            "checkpoint_path": self.ckpt_path,
        }
        base.update(info)
        meta = self.image_dir / "metadata.json"
        if meta.exists():
            try:
                existing = json.loads(meta.read_text())
                existing.update(base)
                base = existing
            except Exception:
                pass
        meta.write_text(json.dumps(base, indent=2, default=str))
        print(f"  Metadata: {meta}")


# ── 17.0b Alignment-MLP helper ───────────────────────────────────────
import torch.nn as nn
import torch.nn.functional as F

class _TextAligner(nn.Module):
    def __init__(self, in_dim=512, hidden=768, out_dim=1280):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(in_dim, hidden), nn.GELU(), nn.Dropout(0.1),
            nn.Linear(hidden, out_dim),
        )
    def forward(self, x):
        return self.net(x)


def inject_alignment_mlp(pipeline, model, dataset_obj, val_idx,
                         concepts_dict, class_concepts_map, n_classes,
                        epochs=800, batch_size=32, lambda_push=0.0, push_margin=0.6):
    model.eval()
    loader = DataLoader(Subset(dataset_obj, val_idx),
                        batch_size=batch_size, shuffle=False, num_workers=0)
    feats, labels = [], []
    with torch.no_grad():
        for images, lab in loader:
            images = images.to(device)
            f = F.normalize(model(images)[0], dim=-1)
            feats.append(f.cpu()); labels.append(lab)
    feats  = torch.cat(feats)
    labels = torch.cat(labels)

    protos = {}
    for c in range(n_classes):
        mask = (labels == c)
        if mask.sum() < 2:
            print(f"  WARN: class {c} has {int(mask.sum())} samples")
            continue
        protos[c] = F.normalize(feats[mask].median(0).values, dim=0).to(device)

    biomedclip.eval()
    with torch.no_grad():
        tokens    = tokenizer(concepts_dict["concepts"]).to(device)
        text_vecs = F.normalize(biomedclip.encode_text(tokens).float(), dim=-1)

    n_concepts = len(concepts_dict["concepts"])
    targets    = torch.zeros(n_concepts, 1280, device=device)
    for class_id, idxs in class_concepts_map.items():
        if class_id not in protos:
            continue
        for j in idxs:
            if j < n_concepts:
                targets[j] = protos[class_id]

    norms      = targets.norm(dim=-1, keepdim=True).clamp(min=1e-8)
    targets    = targets / norms
    has_target = targets.norm(dim=-1) > 0.5
    missing    = (~has_target).sum().item()
    if missing:
        print(f"  WARN: {missing} concept(s) unmapped")

    # same-class concept index pairs for the push-apart term
    pair_i, pair_j = [], []
    for _cls, _idxs in class_concepts_map.items():
        for _a in range(len(_idxs)):
            for _b in range(_a + 1, len(_idxs)):
                if _idxs[_a] < n_concepts and _idxs[_b] < n_concepts:
                    pair_i.append(_idxs[_a]); pair_j.append(_idxs[_b])
    pair_i = torch.tensor(pair_i, device=device, dtype=torch.long)
    pair_j = torch.tensor(pair_j, device=device, dtype=torch.long)

    aligner = _TextAligner().to(device)
    opt     = torch.optim.Adam(aligner.parameters(), lr=1e-3, weight_decay=1e-4)
    aligner.train()
    for ep in range(epochs):
        opt.zero_grad()
        proj = F.normalize(aligner(text_vecs), dim=-1)
        pull = (1 - (proj * targets).sum(dim=-1))[has_target].mean()
        if lambda_push > 0 and len(pair_i) > 0:
            pair_cos = (proj[pair_i] * proj[pair_j]).sum(dim=-1)
            push = torch.clamp(pair_cos - push_margin, min=0).mean()  # 0 once separated past margin
        else:
            push = torch.zeros((), device=device)
        loss = pull + lambda_push * push
        loss.backward()
        opt.step()
        if (ep + 1) % 200 == 0:
            print(f"    epoch {ep+1:4d}  pull={pull.item():.5f}  "
                  f"push={float(push):+.5f}  total={loss.item():.5f}")

    aligner.eval()
    with torch.no_grad():
        aligned_vecs = F.normalize(aligner(text_vecs), dim=-1)

    pipeline.concept_explainer.embeddings = aligned_vecs
    print(f"  Injected aligned vectors {tuple(aligned_vecs.shape)} into pipeline")
    return aligner, protos, aligned_vecs


BUSI_CLASS_CONCEPTS = {
    0: [0, 1],
    1: [2, 3, 4, 5],
    2: [6, 7, 8, 9],
}

### 17.1b Concept separation: push-apart alignment (Option 1)

The baseline alignment loss pulls every within-class concept to the **same** class prototype, so the four Malignant (and four Benign) concept vectors collapse onto one point. That is why every Malignant concept scores ~0.946 on a Malignant image (Table 9) and why their LayerCAM heatmaps are visually identical: the concept vector is the gradient target, so identical vectors give identical maps.

Adding a within-class repulsion term spreads the vectors apart while still keeping each near its class anchor:

`L_align = pull_to_prototype + lambda_push * mean_{same-class i,j} cos(M(text_i), M(text_j))`

The cell below sweeps `lambda_push`, tracks two things, and lets you pick a value:
- **within-class concept-to-concept cosine** (lower = more separated), and
- **mean class-defining score** (must stay high, or we have broken class discrimination to buy separation).

What this fixes and what it does not: separating the vectors makes the per-concept scores and heatmaps distinct again, which directly answers the contradictory-ranking and identical-heatmap complaints. It does **not** by itself guarantee the correct-grade concept wins per image on Mild/Moderate DR. That requires the per-image contrastive loss (Option 2). After picking a `lambda_push`, re-run the 17.6 per-image WHERE eval to see how much of the per-image collapse this actually buys.

# ── 17.1b  Push-apart lambda sweep (BUSI) ────────────────────────────
import numpy as np, pandas as pd, torch
import torch.nn.functional as F
from torch.utils.data import DataLoader, Subset

BUSI_CONCEPT_NAMES = BUSI_CONCEPTS["concepts"]
VAL_IDX_BUSI       = folds_busi[0][1]
CLASS_NAMES        = ["Normal", "Benign", "Malignant"]

def _within_class_cc_cosine(vecs):
    out = {}
    for cls, idxs in BUSI_CLASS_CONCEPTS.items():
        if len(idxs) < 2:
            continue
        sims = [(vecs[idxs[a]] * vecs[idxs[b]]).sum().item()
                for a in range(len(idxs)) for b in range(a + 1, len(idxs))]
        out[cls] = float(np.mean(sims))
    return out

def _concept_table(vecs):
    explain_model_g01.eval()
    loader = DataLoader(Subset(full_busi_ds, VAL_IDX_BUSI),
                        batch_size=16, shuffle=False, num_workers=0)
    rows = []
    with torch.no_grad():
        for imgs, labs in loader:
            feats = F.normalize(explain_model_g01(imgs.to(device))[0], dim=-1)
            sims  = feats @ vecs.to(device).T
            for i in range(feats.shape[0]):
                row = {"true": int(labs[i])}
                row.update({BUSI_CONCEPT_NAMES[j]: sims[i, j].item()
                            for j in range(len(BUSI_CONCEPT_NAMES))})
                rows.append(row)
    return pd.DataFrame(rows).groupby("true").mean()

push_vecs = {}
rows = []
for lam in [0.0, 0.3, 0.5, 1.0]:
    print(f"\n=== lambda_push = {lam} ===")
    _, _, vecs = inject_alignment_mlp(
        pipeline           = busi_pipeline_aligned_g01,
        model              = explain_model_g01,
        dataset_obj        = full_busi_ds,
        val_idx            = VAL_IDX_BUSI,
        concepts_dict      = BUSI_CONCEPTS,
        class_concepts_map = BUSI_CLASS_CONCEPTS,
        n_classes          = cfg.busi_n_classes,
        lambda_push        = lam,
    )
    vecs = vecs.detach().cpu()
    push_vecs[lam] = vecs
    cc  = _within_class_cc_cosine(vecs)
    tbl = _concept_table(vecs)
    diag = [tbl.loc[ci, [BUSI_CONCEPT_NAMES[j] for j in BUSI_CLASS_CONCEPTS[ci]]].mean()
            for ci in range(len(CLASS_NAMES))]
    rows.append({
        "lambda_push":            lam,
        "cc-cos Benign":          round(cc.get(1, float("nan")), 3),
        "cc-cos Malignant":       round(cc.get(2, float("nan")), 3),
        "class-defining score":   round(float(np.mean(diag)), 3),
    })

print("\nlower cc-cos = more separated;  class-defining score must stay high")
print(pd.DataFrame(rows).to_string(index=False))

def load_all_masks(image_path, size=(300, 300)):
    """Union every mask file for this image: _mask, _mask_1, _mask_2, ..."""
    base, ext = os.path.splitext(image_path)
    paths = [f"{base}_mask{ext}"] + sorted(glob.glob(f"{base}_mask_*{ext}"))
    union = None
    for p in paths:
        if os.path.isfile(p):
            m = np.array(PILImage.open(p).convert("L").resize(size, PILImage.NEAREST))
            m = (m > 127).astype(np.uint8)
            union = m if union is None else np.maximum(union, m)
    return union

def us_mask_of(img_np):
    from explainability import _ultrasound_mask
    return (_ultrasound_mask(img_np) > 0.5).astype(np.uint8)

        if lesion is None:

        if glob.glob(f"{base}_mask_*{ext}"):

# ── Helpers ──────────────────────────────────────────────────────
def load_lesion_mask(image_path):
    """Try standard BUSI mask filename pattern."""
    p = Path(image_path)
    stem = p.stem  # e.g. "malignant (84)"
    parent = p.parent
    candidates = [
        parent / f"{stem}_mask.png",
        parent / f"{stem}_mask_1.png",
        parent / f"{stem} mask.png",
    ]
    for c in candidates:
        if c.exists():
            m = np.array(PILImage.open(c).convert("L").resize((300, 300)))
            return (m > 127).astype(np.uint8)
    return None

def apply_margin_suppression(arr, frac=MARGIN_FRAC):
    """Zero out a margin around the array borders."""
    arr = arr.copy()
    h, w = arr.shape[:2]
    mh, mw = int(h * frac), int(w * frac)
    arr[:mh, :] = 0
    arr[-mh:, :] = 0
    arr[:, :mw] = 0
    arr[:, -mw:] = 0
    return arr

def compute_metrics_for_class(class_idx, class_name):
    """Return dict of metrics + list of per-image diagnostic data."""
    pipe = busi_pipeline_aligned_g01
    pipe.concept_explainer.embeddings = aligned_vecs_busi_g01

    # Filter validation images by class
    items_in_class = []
    for global_idx in VAL_IDX_BUSI:
        image_path, label = full_busi_ds.items[global_idx]
        if label == class_idx:
            items_in_class.append((global_idx, image_path))

    if not items_in_class:
        return None, []

    pointing_in_lesion = []
    pointing_in_us     = []
    iou_top20          = []
    diag_records       = []  # for the figure

    for global_idx, image_path in items_in_class:
        # Image
        img_tensor, _ = full_busi_ds[global_idx]
        img_tensor = img_tensor.unsqueeze(0).to(device)
        raw = PILImage.open(image_path).convert("RGB").resize((300, 300))
        img_np = np.array(raw).astype(np.float32) / 255.0

        # Heatmap (use the predicted-class CAM, i.e. the standard explanation)
        result = pipe.explain(img_tensor)
        cam = result["heatmap"]                # (300, 300), already pipeline-normalised
        cam = apply_margin_suppression(cam)

        # Ultrasound region mask
        us_mask = _ultrasound_mask(img_np)
        us_mask = apply_margin_suppression(us_mask.astype(np.uint8))

        # Lesion mask (Benign / Malignant only)
        lesion_mask = load_lesion_mask(image_path)
        if lesion_mask is not None:
            lesion_mask = apply_margin_suppression(lesion_mask)

        # Argmax inside the (margin-suppressed) ultrasound region
        cam_in_us = cam * us_mask
        if cam_in_us.max() <= 0:
            continue
        flat_idx = np.argmax(cam_in_us)
        ay, ax = np.unravel_index(flat_idx, cam.shape)

        in_us = us_mask[ay, ax] == 1
        pointing_in_us.append(in_us)

        if lesion_mask is not None:
            in_lesion = lesion_mask[ay, ax] == 1
            pointing_in_lesion.append(in_lesion)

            # IoU at top-20%
            thresh = np.quantile(cam[us_mask == 1], 1 - TOP_PCT) if (us_mask == 1).sum() > 0 else 1.0
            heatmap_bin = (cam >= thresh).astype(np.uint8) * us_mask
            inter = ((heatmap_bin == 1) & (lesion_mask == 1)).sum()
            union = ((heatmap_bin == 1) | (lesion_mask == 1)).sum()
            iou = inter / union if union > 0 else 0.0
            iou_top20.append(iou)

        diag_records.append({
            "img_np":      img_np,
            "cam":         cam,
            "lesion_mask": lesion_mask,
            "us_mask":     us_mask,
            "argmax":      (ay, ax),
            "image_path":  str(image_path),
        })

    metrics = {
        "class":             class_name,
        "n":                 len(items_in_class),
        "n_with_lesion":     len(pointing_in_lesion),
        "pointing_in_lesion": float(np.mean(pointing_in_lesion)) if pointing_in_lesion else None,
        "pointing_in_us":    float(np.mean(pointing_in_us)) if pointing_in_us else None,
        "mean_iou_top20":    float(np.mean(iou_top20)) if iou_top20 else None,
        "std_iou_top20":     float(np.std(iou_top20)) if iou_top20 else None,
    }
    return metrics, diag_records

# ── Diagnostic figure (top row: img + lesion outline + argmax cross) ──
def save_diagnostic(diag_records, class_name, out_path, n_show=6):
    diag_records = diag_records[:n_show]
    n = len(diag_records)
    if n == 0:
        return
    fig, axes = plt.subplots(2, n, figsize=(3.0 * n, 6.0))
    if n == 1:
        axes = axes.reshape(2, 1)

        img      = rec["img_np"]
        cam      = rec["cam"]
        lesion   = rec["lesion_mask"]
        ay, ax   = rec["argmax"]
        img_uint = (img * 255).astype(np.uint8)

        # Top: original + lesion outline (green) + argmax (red cross)
        axes[0, col].imshow(img_uint)
        if lesion is not None:
            axes[0, col].contour(lesion, levels=[0.5], colors="lime", linewidths=1.5)
        axes[0, col].plot(ax, ay, "r+", markersize=18, markeredgewidth=2.5)
        axes[0, col].set_xticks([]); axes[0, col].set_yticks([])
        axes[0, col].set_title(Path(rec["image_path"]).stem, fontsize=9)

        # Bottom: heatmap + lesion outline (white) + argmax (black cross)
        axes[1, col].imshow(img_uint)
        # soft-mask the heatmap for display
        soft = gaussian_filter(rec["us_mask"].astype(np.float32), sigma=9)
        if soft.max() > 0:
            soft = soft / soft.max()
        h_disp = cam * soft
        if h_disp.max() > h_disp.min():
            h_disp = (h_disp - h_disp.min()) / (h_disp.max() - h_disp.min() + 1e-8)
        axes[1, col].imshow(h_disp, cmap="jet", alpha=0.45, vmin=0, vmax=1)
        if lesion is not None:

        axes[1, col].plot(ax, ay, "kx", markersize=14, markeredgewidth=2.5)
        axes[1, col].set_xticks([]); axes[1, col].set_yticks([])

    # Row labels
    axes[0, 0].set_ylabel("Original\n(green=lesion, red=argmax)", fontsize=10)
    axes[1, 0].set_ylabel("Heatmap\n(white=lesion, black=argmax)", fontsize=10)
    plt.suptitle(f"BUSI localisation diagnostic — {class_name} (γ=0.1)",

# ── 17.5d  Multi-prototype alignment for DR ──────────────────────────
import numpy as np, torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import DataLoader, Subset
from sklearn.cluster import KMeans

K_SUB        = 2          # sub-clusters per grade (1 = reduces to single-prototype)
ALIGN_EPOCHS = 800
DR_GRADES    = ["No DR","Mild","Moderate","Severe","PDR"]
VAL_IDX_DR   = folds_dr[0][1]
_dr_ds       = full_dr_ds if "full_dr_ds" in dir() else dr_full_ds

# 1) collect val embeddings + labels once
explain_model_dr_g01.eval()
emb, lab = [], []
loader = DataLoader(Subset(_dr_ds, VAL_IDX_DR), batch_size=32, shuffle=False, num_workers=0)
with torch.no_grad():
    for imgs, labs in loader:
        if imgs.shape[1] == 1:
            imgs = imgs.repeat(1, 3, 1, 1)
        f = F.normalize(explain_model_dr_g01(imgs.to(device))[0], dim=-1).cpu()
        emb.append(f); lab.append(labs)
emb = torch.cat(emb); lab = torch.cat(lab).numpy()
print(f"val embeddings: {tuple(emb.shape)}")

# 2) build up to K_SUB sub-prototypes per grade
sub_protos = {}
for g in range(5):
    fg = emb[lab == g].numpy()
    if len(fg) < K_SUB:
        ctr = [fg.mean(0)] if len(fg) else []
    else:
        km  = KMeans(n_clusters=K_SUB, n_init=10, random_state=0).fit(fg)
        ctr = [km.cluster_centers_[k] for k in range(K_SUB)]
    sub_protos[g] = [F.normalize(torch.tensor(c, dtype=torch.float32), dim=0).to(device) for c in ctr]
    print(f"  grade {g} ({DR_GRADES[g]:<9s}): {len(fg):>5d} imgs -> {len(ctr)} sub-prototype(s)")

# 3) per-concept target = densest sub-prototype of its grade
biomedclip.eval()
with torch.no_grad():
    tok  = tokenizer(DR_CONCEPTS["concepts"]).to(device)
    tvec = F.normalize(biomedclip.encode_text(tok).float(), dim=-1)
n_con = len(DR_CONCEPTS["concepts"])

grade_target = {}
for g, protos in sub_protos.items():
    if len(protos) == 1:
        grade_target[g] = protos[0]; continue
    fg = emb[lab == g].to(device)
    P  = torch.stack(protos)
    nearest = (fg @ P.T).argmax(dim=1)
    counts  = torch.bincount(nearest, minlength=len(protos))
    grade_target[g] = protos[int(counts.argmax())]

targets = torch.zeros(n_con, 1280, device=device)
has_t   = torch.zeros(n_con, dtype=torch.bool, device=device)
for g, idxs in CLASS_CONCEPTS_DR.items():
    for j in idxs:
        if j < n_con:
            targets[j] = grade_target[g]; has_t[j] = True

# 4) refit the alignment MLP
class _Aln(nn.Module):
    def __init__(s):
        super().__init__()
        s.net = nn.Sequential(nn.Linear(512,768), nn.GELU(), nn.Dropout(0.1), nn.Linear(768,1280))
    def forward(s,x): return s.net(x)

aln = _Aln().to(device)
opt = torch.optim.Adam(aln.parameters(), lr=1e-3, weight_decay=1e-4)
for ep in range(ALIGN_EPOCHS):
    opt.zero_grad()
    proj = F.normalize(aln(tvec), dim=-1)
    loss = (1 - (proj * targets).sum(-1))[has_t].mean()
    loss.backward(); opt.step()
    if (ep+1) % 200 == 0:
        print(f"    epoch {ep+1:4d}  align_loss={loss.item():.5f}")

aln.eval()
with torch.no_grad():
    aligned_vecs_dr_MULTI = F.normalize(aln(tvec), dim=-1).detach()
print("Built: aligned_vecs_dr_MULTI")

def ultrasound_region_mask(img_np):
    """Binary mask of the actual scan area, with 5% edge suppression."""
    if img_np.dtype != np.uint8:
        gray = (img_np.mean(axis=2) if img_np.ndim == 3 else img_np)
    else:
        gray = (img_np.astype(np.float32).mean(axis=2) / 255.0
                if img_np.ndim == 3 else img_np.astype(np.float32) / 255.0)
    region = (gray > 0.03).astype(np.uint8)
    margin = max(1, int(0.05 * min(region.shape)))
    eroded = binary_erosion(region, iterations=margin)
    return eroded.astype(np.uint8)

def load_busi_mask(image_path):
    """BUSI lesion mask: foo.png -> foo_mask.png next to it."""
    base, ext = os.path.splitext(image_path)
    mask_path = f"{base}_mask{ext}"
    if not os.path.isfile(mask_path):
        return None
    m = Image.open(mask_path).convert("L").resize((300, 300), Image.NEAREST)
    return (np.array(m) > 127).astype(np.uint8)

def iter_samples():
    """Walk the validation set, sample N_PER_CLS per class."""
    counts = defaultdict(int)

    if DATASET == "busi":
        # Use the dataset's own items list filtered by fold-1 val indices

            class_name = ["Normal", "Benign", "Malignant"][label]
            if counts[class_name] >= N_PER_CLS:
                continue
            counts[class_name] += 1

            # Load image via the dataset (uses the same transform model was trained on)
            img_tensor, _ = full_ds[global_idx]
            img_tensor = img_tensor.unsqueeze(0).to(device)

            # Raw image array for masking + display
            raw = Image.open(image_path).convert("RGB").resize((300, 300))
            img_np = np.array(raw)

            lesion_mask = load_busi_mask(image_path)
            yield img_tensor, img_np, lesion_mask, class_name, os.path.basename(image_path)

            class_name = class_names[label]

            counts[class_name] += 1

            if x.shape[1] == 1:

def safe_name(s):
    """Sanitise a concept name into a filename-safe slug."""
    s = s.lower().replace(" ", "_").replace("/", "_")
    return re.sub(r"[^a-z0-9_]", "", s)

def soft_mask_for_display(img_np, dataset):
    """Same soft-mask logic that fixed your diagnostic figures."""
    if dataset == "busi":
        hard = _ultrasound_mask(img_np)
    elif dataset == "dr":
        hard = _fundus_mask(img_np)
    else:
        return None
    soft = gaussian_filter(hard.astype(np.float32), sigma=VIS_MASK_SIGMA)
    if soft.max() > 0:
        soft = soft / soft.max()
    return soft

def save_original(img_np, out_path):
    """Save the original image (no overlay)."""
    img_uint8 = (img_np * 255).astype(np.uint8) if img_np.dtype != np.uint8 else img_np
    fig, ax = plt.subplots(figsize=(5, 5))
    ax.imshow(img_uint8)
    ax.axis("off")
    fig.tight_layout(pad=0)
    fig.savefig(out_path, dpi=140, bbox_inches="tight", pad_inches=0)
    plt.close(fig)

def save_panel(img_np, heatmap, soft_mask, concept_name, score,
               out_path, blind=False):
    """Save one concept panel.
       If blind=True, no concept name and no score are visible."""
    if soft_mask is not None:
        h = heatmap * soft_mask
    else:
        h = heatmap.copy()
    if h.max() > h.min():
        h = (h - h.min()) / (h.max() - h.min() + 1e-8)

    img_uint8 = (img_np * 255).astype(np.uint8) if img_np.dtype != np.uint8 else img_np

    fig, ax = plt.subplots(figsize=(5, 5))
    ax.imshow(img_uint8)
    ax.imshow(h, cmap="jet", alpha=0.45, vmin=0, vmax=1)
    ax.axis("off")
    if not blind:
        ax.set_title(f"{concept_name}\nsim={score:+.2f}", fontsize=11)
    fig.tight_layout(pad=0)
    fig.savefig(out_path, dpi=140, bbox_inches="tight", pad_inches=0.02)
    plt.close(fig)

def save_grid_named(img_np, heatmaps, scores, concepts, soft_mask,
                    out_path, suptitle):
    """Full named grid — used as input for the disease-classification call."""
    n = len(concepts)
    ncols = min(5, n)
    nrows = (n + ncols - 1) // ncols
    img_uint8 = (img_np * 255).astype(np.uint8) if img_np.dtype != np.uint8 else img_np

    fig, axes = plt.subplots(nrows, ncols, figsize=(3.2*ncols, 3.2*nrows))
    axes = np.array(axes).flatten()
    for i, (concept, hmap) in enumerate(zip(concepts, heatmaps)):
        h = hmap * soft_mask if soft_mask is not None else hmap.copy()
        if h.max() > h.min():
            h = (h - h.min()) / (h.max() - h.min() + 1e-8)
        axes[i].imshow(img_uint8)
        axes[i].imshow(h, cmap="jet", alpha=0.45, vmin=0, vmax=1)
        axes[i].set_title(f"{concept}\nsim={scores[concept]:+.2f}", fontsize=9)
        axes[i].axis("off")
    for j in range(n, len(axes)):
        axes[j].axis("off")

    plt.suptitle(suptitle, fontsize=12, fontweight="bold")
    plt.tight_layout()
    plt.savefig(out_path, dpi=130, bbox_inches="tight")
    plt.close(fig)

def generate_cases(dataset, pipe, aligned_vecs, explain_model, full_ds,
                   val_idx, class_names, n_per_class, out_root):
    pipe.concept_explainer.embeddings = aligned_vecs
    concepts = pipe.concept_explainer.concepts

    cases_done = []
    counts = {c: 0 for c in class_names}
    case_counter = 0

    if dataset == "busi":
        # iterate using full_ds.items (we have file paths)
        iterator = []
        for global_idx in val_idx:
            image_path, label = full_ds.items[global_idx]
            class_name = class_names[label]
            if counts[class_name] >= n_per_class:
                continue
            counts[class_name] += 1
            iterator.append((global_idx, image_path, class_name))
            if all(counts[c] >= n_per_class for c in class_names):
                break

    else:  # dr
        loader = DataLoader(Subset(full_ds, val_idx), batch_size=1, shuffle=False)
        iterator = []
        for i, (x, y) in enumerate(loader):
            label = int(y.item())
            class_name = class_names[label]
            if counts[class_name] >= n_per_class:
                continue
            counts[class_name] += 1
            iterator.append((i, x, class_name))
            if all(counts[c] >= n_per_class for c in class_names):
                break

    print(f"\n[{dataset.upper()}] {len(iterator)} cases to generate")

    for entry in iterator:
        case_counter += 1
        if dataset == "busi":
            global_idx, image_path, class_name = entry
            img_id = os.path.splitext(os.path.basename(image_path))[0].replace(" ", "_")
            img_tensor, _ = full_ds[global_idx]
            img_tensor = img_tensor.unsqueeze(0).to(device)
            raw = Image.open(image_path).convert("RGB").resize((300, 300))
            img_np = np.array(raw).astype(np.float32) / 255.0
        else:
            i, x, class_name = entry
            if x.shape[1] == 1:
                x = x.repeat(1, 3, 1, 1)
            img_tensor = x.to(device)
            img_np = x.squeeze().permute(1, 2, 0).cpu().numpy()
            if img_np.shape[-1] == 1:
                img_np = np.repeat(img_np, 3, axis=-1)
            img_id = f"dr_idx{i:04d}"

        case_slug = f"case_{case_counter:03d}__{safe_name(class_name)}__{safe_name(img_id)}"
        case_dir = out_root / dataset / case_slug
        (case_dir / "named").mkdir(parents=True, exist_ok=True)

            "image_id":   img_id,
            "n_panels":   len(concepts),
            "case_dir":   str(case_dir.relative_to(out_root.parent.parent)),
        })

        print(f"  [{case_counter:3d}] {class_name:<14s}  {img_id:<22s}  "
              f"-> {case_slug}")

    return cases_done

import numpy as np, torch, json, os
import torch.nn.functional as F
from torch.utils.data import DataLoader, Subset

def per_concept_stats(model, ds, val_idx, concepts, class_concepts, vecs):
    model.eval()
    V = vecs.to(device); Cn = V.shape[0]
    top1_hits=np.zeros(Cn); grade_imgs=np.zeros(Cn)
    sim_sum=np.zeros(Cn); sim_n=np.zeros(Cn); rank_sum=np.zeros(Cn)
    cgrade={}
    for g,idxs in class_concepts.items():
        for j in idxs: cgrade[j]=g
    ld=DataLoader(Subset(ds,val_idx),batch_size=32,shuffle=False,num_workers=0)
    with torch.no_grad():
        for imgs,labs in ld:
            if imgs.shape[1]==1: imgs=imgs.repeat(1,3,1,1)
            f=F.normalize(model(imgs.to(device))[0],dim=-1)
            sims=(f@V.T).cpu().numpy(); top1=sims.argmax(1); order=(-sims).argsort(1)
            for bi,lab in enumerate(labs.numpy().tolist()):
                for j in class_concepts[lab]:
                    grade_imgs[j]+=1; sim_sum[j]+=sims[bi,j]; sim_n[j]+=1
                    rank_sum[j]+=int(np.where(order[bi]==j)[0][0])+1
                    if top1[bi]==j: top1_hits[j]+=1
    rows=[]
    for j in range(Cn):
        if grade_imgs[j]>0:
            rows.append(dict(concept=concepts[j], grade=cgrade.get(j,-1),
                n=int(grade_imgs[j]), top1=round(top1_hits[j]/grade_imgs[j],3),
                mean_sim=round(sim_sum[j]/sim_n[j],3), mean_rank=round(rank_sum[j]/sim_n[j],2)))
    return rows

# == 19.3  Concept-confusion matrix (which concept wins, by true grade) ==
import numpy as np, torch, matplotlib.pyplot as plt
import torch.nn.functional as F

def concept_confusion(model, ds, val_idx, n_classes, vecs):
    model.eval(); V=vecs.to(device); Cn=V.shape[0]
    M=np.zeros((n_classes,Cn))
    ld=DataLoader(Subset(ds,val_idx),batch_size=32,shuffle=False,num_workers=0)
    with torch.no_grad():
        for imgs,labs in ld:
            if imgs.shape[1]==1: imgs=imgs.repeat(1,3,1,1)
            f=F.normalize(model(imgs.to(device))[0],dim=-1)
            t1=(f@V.T).argmax(1).cpu().numpy()
            for lab,t in zip(labs.numpy(),t1): M[int(lab),t]+=1
    return M/M.sum(1,keepdims=True).clip(min=1)

# ── 19.6  MULTI-FOLD per-concept correctness (mean ± std) ────────────
import numpy as np, torch, json, os, glob
import torch.nn.functional as F
from torch.utils.data import DataLoader, Subset
os.makedirs("paper_figures_FINAL/quant", exist_ok=True)
_dr_ds = full_dr_ds if 'full_dr_ds' in dir() else dr_full_ds

def _per_concept_fold(model, ds, val_idx, class_concepts, vecs, n_concepts):
    model.eval(); V=vecs.to(device)
    hits=np.zeros(n_concepts); nn=np.zeros(n_concepts); rank=np.zeros(n_concepts)
    ld=DataLoader(Subset(ds,val_idx),batch_size=32,shuffle=False,num_workers=0)
    with torch.no_grad():
        for x,y in ld:
            if x.shape[1]==1: x=x.repeat(1,3,1,1)
            f=F.normalize(model(x.to(device))[0],dim=-1)
            sims=(f@V.T).cpu().numpy(); t1=sims.argmax(1); order=(-sims).argsort(1)
            for bi,lab in enumerate(y.numpy().tolist()):
                for j in class_concepts[lab]:
                    nn[j]+=1; rank[j]+=int(np.where(order[bi]==j)[0][0])+1
                    if t1[bi]==j: hits[j]+=1
    t=np.divide(hits,nn,out=np.full(n_concepts,np.nan),where=nn>0)
    r=np.divide(rank,nn,out=np.full(n_concepts,np.nan),where=nn>0)
    return t,r

def _run_multifold(name, ckpt_glob, build_model, ds, concepts, class_concepts,
                   inject_fn, concepts_dict, n_classes, n_folds, folds):
    nC=len(concepts); T=[]; R=[]
    for k in range(n_folds):
        ckpt=f"checkpoints/{ckpt_glob}/fold_{k+1}_best.pt"
        if not os.path.isfile(ckpt):
            print(f"  {name} fold {k+1}: MISSING, skip"); continue
        model=build_model(ckpt)
        # rebuild alignment on THIS fold's val split
        _,_,vecs = inject_fn(model, ds, folds[k][1], concepts_dict, class_concepts, n_classes)
        t,r=_per_concept_fold(model, ds, folds[k][1], class_concepts, vecs.detach().cpu(), nC)
        T.append(t); R.append(r); print(f"  {name} fold {k+1}: done")
    T=np.vstack(T); R=np.vstack(R)

def overlay_png_b64(img_np, heatmap):
    us = _ultrasound_mask(img_np); hm = heatmap*us
    if hm.max()>hm.min(): hm=(hm-hm.min())/(hm.max()-hm.min()+1e-8)
    over = LayerCAM.overlay(img_np, hm, alpha=0.5)
    buf=io.BytesIO(); PILImage.fromarray(over).save(buf,format="PNG")
    return base64.b64encode(buf.getvalue()).decode()
